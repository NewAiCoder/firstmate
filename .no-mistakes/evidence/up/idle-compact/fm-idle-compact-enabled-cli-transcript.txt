$ FM_ROOT_OVERRIDE=$LAB bash bin/fm-idle-compact.sh enabled   # no config/idle-compact present
exit=1

$ : > $LAB/config/idle-compact; ... enabled   # present but empty -> default 15
15
exit=0

$ printf "42\n" > $LAB/config/idle-compact; ... enabled   # explicit minutes
42
exit=0

$ printf "banana\n" > $LAB/config/idle-compact; ... enabled   # invalid value treated as disabled (typo-safe)
exit=1

$ printf "0\n" > $LAB/config/idle-compact; ... enabled   # zero treated as disabled
exit=1
